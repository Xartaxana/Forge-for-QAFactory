"""CL4-1 N2 differential compare (scratch, never committed). usage: python -B d_compare.py <old.jsonl> <new.jsonl>
Exit 0 only when D1 identical and D2 (i)-(iii) hold; (iv) is d_problems.py. Prints every delta row."""
import json
import sys

old = [json.loads(x) for x in open(sys.argv[1], encoding="utf-8")]
new = [json.loads(x) for x in open(sys.argv[2], encoding="utf-8")]
bad = 0
if old[0]["path"] == new[0]["path"] or old[1]["path"] == new[1]["path"]:
    print("VOID: old and new loaded the same file"); sys.exit(3)
assert len(old) == len(new), (len(old), len(new))
d1 = d2 = 0
for o, n in zip(old[2:], new[2:]):
    assert o["id"] == n["id"] and o["d"] == n["d"]
    if o["d"] == "D1":
        d1 += 1
        if o != n:
            bad += 1; print("D1 DIFF", o["id"], o, n)
        continue
    d2 += 1
    if o == n:
        continue
    print("D2 delta:", o["id"], "| gate %s->%s | why %r -> %r | word %r -> %r" % (o["gate"], n["gate"], o["why"], n["why"], o["word"], n["word"]))
    if o["stalled_is_bool"]:
        bad += 1; print("  VIOLATES (i): stalled is a bool")
    if o["gate"] != n["gate"] and not (o["gate"] is False and n["gate"] is True):
        bad += 1; print("  VIOLATES (ii): gate delta is not False->True")
    if not o["stalled_is_bool"]:
        pass
print("D1 rows %d, D2 rows %d, violations %d" % (d1, d2, bad))
