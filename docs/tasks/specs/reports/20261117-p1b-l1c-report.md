"""L1c census catch-up: rows for every 'no row names it' joint of the --check output. Usage: l1c.py <worktree> <check-output> [--write]"""
import json, re, sys, collections
from pathlib import Path
wt, chk = Path(sys.argv[1]), Path(sys.argv[2]); write = "--write" in sys.argv
pkg = wt / "ai-packages" / "qa-factory"
sys.path.insert(0, str(pkg / "scripts"))
import null_outcomes as O, stage_joints as J
joints = {j["check"]: j for j in O.load_data(J.DATA)[0]["joints"]}
EXITS = pkg / "scripts" / "tests" / "fixtures" / "driver-exits"
GROUPS = {"S0": "s0", **{f"S{i}": "s1-s4" for i in (1, 2, 3, 4)}, "S5": "s5", **{f"S{i}": "s6-s9" for i in (6, 7, 8, 9)}}
def group(jid):
    st = joints[jid].get("stages")
    if not st: raise SystemExit(f"STOP no stages: {jid}")
    if st == ["*"]: return "gates"  # Lead decision 17.11: '*' = every stage -> gates
    gs = set()
    for s in st:
        if s not in GROUPS: raise SystemExit(f"STOP stage {s} outside S0-S9: {jid}")
        gs.add(GROUPS[s])
    return gs.pop() if len(gs) == 1 else "gates"
files = {p.stem: p for p in EXITS.glob("*.json") if p.name not in ("classes.json", "known_red.json")}
data = {k: json.loads(p.read_text(encoding="utf-8")) for k, p in files.items()}
# step 1: group rule on standing joints
bad = []; n = 0
for g, d in data.items():
    if g == "hooks": continue
    for jid in d["joints"]:
        if jid in joints:
            n += 1
            if group(jid) != g and jid != "branch_accounting:*": bad.append((jid, g, group(jid)))
        elif ":*" not in jid: bad.append((jid, g, "not in stage-joints"))
print("standing joints checked:", n, "mismatches:", bad, "(standing exception: branch_accounting:* in s1-s4, rule gives gates - left to the group walk node)")
if bad: raise SystemExit("STOP mismatch")
new = [m.group(1) for l in chk.read_text(encoding="utf-8").splitlines() if (m := re.match(r"FAIL joint (.+?): the driver has it and no row names it", l))]
other = [l for l in chk.read_text(encoding="utf-8").splitlines() if l.startswith("FAIL") and "no row names it" not in l]
if other: raise SystemExit(f"STOP other FAIL lines: {other}")
star = collections.Counter(); table = collections.Counter(); added = collections.Counter()
for jid in new:
    g = group(jid); table[(jid.split(":")[0], g)] += 1; star[jid.split(":")[0]] += joints[jid]["stages"] == ["*"]; added[g] += 1
    if write:
        data[g]["joints"][jid] = {"reach": "unwalked", "why_unreached": "not walked yet: joint added on main after N1 (census catch-up L1c); the group's walk node writes its rows"}
        data[g]["ratchet"]["raised_by"].append(f"unwalked | joints/{jid} | L1c census catch-up, 2026-11-17")
for (f, g), c in sorted(table.items(), key=lambda x: (-x[1], x[0])): print(f"{f}\t{g}\t{c}")
print("stages * -> gates:", {k: v for k, v in star.items() if v}, sum(star.values()))
print("per group:", dict(added), "total", len(new))
if write:
    for g, p in files.items(): p.write_text(json.dumps(data[g], indent=1, ensure_ascii=False) + "\n", encoding="utf-8")
