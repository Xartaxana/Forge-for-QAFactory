import ast
import re

SRC = sys.argv[1]
NEG = len(sys.argv) > 2 and sys.argv[2] == "--negative"
text = open(SRC, encoding="utf-8").read()
doc = ast.get_docstring(ast.parse(text), clean=False)
lines = text.splitlines()
# the two comments: section comment above text_head, and the _receipt_core `proved_by` comment line
sec = "\n".join(l for l in lines if l.startswith("# ") and "ONE row judge" in l or l.startswith("# ") and "well enough" in l or l.startswith("# never a verdict"))
rc = next(l for l in lines if 'core["proved_by"] = record["proved_by"]' in l)
blob = doc + "\n" + sec + "\n" + rc

absent = ["or its `fires_as` message is `intended_failure`"]
present = ["cause: judgment", "qa-probe-<token>", "proved_by: critic"]
if NEG:
    present[1] = "qa-probe-<tokne>"   # deliberately misspelt
bad = []
for k in absent:
    if k in blob:
        bad.append("ABSENT key found: " + k)
for k in present:
    if k not in blob:
        bad.append("PRESENT key missing: " + k)
print("FAIL" if bad else "OK", *bad, sep="\n")
sys.exit(1 if bad else 0)
