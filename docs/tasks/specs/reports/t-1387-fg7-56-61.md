    for node in ast.parse(src).body:
        if isinstance(node, ast.FunctionDef) and node.name == symbol:

def run(f, *a):
    try:

    mism = cases = 0

            cases += 1
            if run(old, *args) != run(new, *args):
                mism += 1
    return mism, cases
